function DiscordLib:NewTheme(name, tbl)
    assert(type(name) == "string", "Theme name must be a string")
    assert(type(tbl)  == "table",  "Theme must be a table")
    local base = DefaultTheme()
    for k,v in pairs(tbl) do base[k] = v end
    Themes[name] = base
end

local function MakeDraggable(handle, frame)
    local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
    local function update(input)
        local delta = input.Position - dragStart
        frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
    handle.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
            dragging   = true
            dragStart  = inp.Position
            startPos   = frame.Position
            inp.Changed:Connect(function()
                if inp.UserInputState == Enum.UserInputState.End then dragging = false end
            end)
        end
    end)
    handle.InputChanged:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
            dragInput = inp
        end
    end)
    UserInputService.InputChanged:Connect(function(inp)
        if inp == dragInput and dragging then update(inp) end
    end)
end

local function TouchDrag(element, onMove, onEnd)
    element.InputBegan:Connect(function(inp)
        if inp.UserInputType == Enum.UserInputType.Touch or inp.UserInputType == Enum.UserInputType.MouseButton1 then
            local conn1, conn2
            conn1 = UserInputService.InputChanged:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement then
                    onMove(i)
                end
            end)
            conn2 = UserInputService.InputEnded:Connect(function(i)
                if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
                    if onEnd then onEnd() end
                    conn1:Disconnect()
                    conn2:Disconnect()
                end
            end)
        end
    end)
end

            -- ── IMAGE ───────────────────────────────────────
            function ChannelAPI:Image(opts)
                opts = opts or {}
                local imgSrc   = opts.Image   or ""
                local imgRatio = opts.Ratio   or 16/9
                local imgH     = math.floor((ElementWidth) / imgRatio)

                local Wrap = MakeBaseFrame(imgH + 10)
                Wrap.Parent = ChannelHolder

                local ImgFrame = Instance.new("Frame")
                ImgFrame.Parent           = Wrap
                ImgFrame.BackgroundColor3 = T.Surface
                ImgFrame.BorderSizePixel  = 0
                ImgFrame.Position         = UDim2.new(0, 0, 0, 4)
                ImgFrame.Size             = UDim2.new(1, 0, 0, imgH)
                local IFCorner = Instance.new("UICorner")
                IFCorner.CornerRadius = UDim.new(0, 6)
                IFCorner.Parent       = ImgFrame

                local Img = Instance.new("ImageLabel")
                Img.Name                 = "Img"
                Img.Parent               = ImgFrame
                Img.BackgroundTransparency = 1
                Img.Size                 = UDim2.new(1, 0, 1, 0)
                Img.ScaleType            = Enum.ScaleType.Crop

                -- resolve image source
                local function ResolveImage(src)
                    if src:find("^rbxassetid://") then
                        Img.Image = src
                    elseif src:find("^%d+$") then
                        Img.Image = "rbxassetid://" .. src
                    else
                        -- raw URL (http/https jpg/png) - try getcustomasset, fallback to direct
                        local ok, result = pcall(function()
                            return getcustomasset and getcustomasset(src) or src
                        end)
                        Img.Image = (ok and result) or src
                    end
                end
                ResolveImage(imgSrc)
                RefreshCanvas()

                local ImgAPI = {}
                function ImgAPI:SetImage(src) ResolveImage(src) end
                function ImgAPI:SetTitle(t) end -- no title for image
                function ImgAPI:SetText(t)  end
                function ImgAPI:Destroy()   Wrap:Destroy(); RefreshCanvas() end
                return ImgAPI
            end

            -- ── SEPARATOR ───────────────────────────────────
            function ChannelAPI:Seperator()
                local Wrap = MakeBaseFrame(10)
                Wrap.Parent = ChannelHolder
                local Line = Instance.new("Frame")
                Line.Parent           = Wrap
                Line.BackgroundColor3 = T.Separator
                Line.BorderSizePixel  = 0
                Line.AnchorPoint      = Vector2.new(0, 0.5)
                Line.Position         = UDim2.new(0, 0, 0.5, 0)
                Line.Size             = UDim2.new(1, 0, 0, 1)
                RefreshCanvas()

                local SepAPI = {}
                function SepAPI:Destroy() Wrap:Destroy(); RefreshCanvas() end
                return SepAPI
            end

            -- ── KEYBIND ─────────────────────────────────────
            function ChannelAPI:Bind(opts)
                if type(opts) == "string" then
                    local t, pk, cb = opts, select(2,...), select(3,...)
                    opts = {Text = t, Key = pk, Callback = cb}
                end
                opts = opts or {}
                local bText = opts.Text     or "Keybind"
                local bKey  = (opts.Key and opts.Key.Name) or "None"
                local bCb   = opts.Callback or function() end

                local Wrap = MakeBaseFrame(32)
                Wrap.Parent = ChannelHolder

                local Row = Instance.new("TextButton")
                Row.Parent            = Wrap
                Row.BackgroundColor3  = T.SurfaceAlt
                Row.BorderSizePixel   = 0
                Row.Size              = UDim2.new(1, 0, 0, 30)
                Row.Position          = UDim2.new(0, 0, 0, 2)
                Row.AutoButtonColor   = false
                Row.Text              = ""
                Row.Font              = Enum.Font.Gotham

                local BTitle = Instance.new("TextLabel")
                BTitle.Name               = "BindTitle"
                BTitle.Parent             = Row
                BTitle.BackgroundTransparency = 1
                BTitle.Position           = UDim2.new(0, 8, 0, 0)
                BTitle.Size              = UDim2.new(0.6, 0, 1, 0)
                BTitle.Font              = Enum.Font.Gotham
                BTitle.Text              = bText
                BTitle.TextColor3        = T.TextSecondary
                BTitle.TextSize          = 14
                BTitle.TextXAlignment    = Enum.TextXAlignment.Left

                local BKey = Instance.new("TextLabel")
                BKey.Name               = "BindKey"
                BKey.Parent             = Row
                BKey.BackgroundTransparency = 1
                BKey.AnchorPoint        = Vector2.new(1, 0.5)
                BKey.Position           = UDim2.new(1, -8, 0.5, 0)
                BKey.Size               = UDim2.new(0, 100, 0, 22)
                BKey.Font               = Enum.Font.Gotham
                BKey.Text               = "[" .. bKey .. "]"
                BKey.TextColor3         = T.Accent
                BKey.TextSize           = 13
                BKey.TextXAlignment     = Enum.TextXAlignment.Right

                local listening = false
                Row.MouseButton1Click:Connect(function()
                    if listening then return end
                    listening = true
                    BKey.Text     = "[...]"
                    BKey.TextColor3 = T.TextSecondary
                    local conn
                    conn = UserInputService.InputBegan:Connect(function(inp, gp)
                        if gp then return end
                        if inp.KeyCode ~= Enum.KeyCode.Unknown then
                            bKey  = inp.KeyCode.Name
                            BKey.Text     = "[" .. bKey .. "]"
                            BKey.TextColor3 = T.Accent
                            listening = false
                            conn:Disconnect()
                        end
                    end)
                end)
                UserInputService.InputBegan:Connect(function(inp, gp)
                    if gp then return end
                    if not listening and inp.KeyCode.Name == bKey then
                        pcall(bCb)
                    end
                end)
                RefreshCanvas()

                local BindAPI = {}
                function BindAPI:SetTitle(t) BTitle.Text = t end
                function BindAPI:SetText(t)  BTitle.Text = t end
                function BindAPI:Destroy()   Wrap:Destroy(); RefreshCanvas() end
                return BindAPI
            end

            return ChannelAPI
        end -- Channel

        return ServerAPI
    end -- Server

    return WindowAPI
end -- Window

-- RETURN
